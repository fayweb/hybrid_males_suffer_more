> # Test Sex × HI and Sex × He interactions for each pathway
> test_pathway_interactions <- function(data, pathways) {
+   results <- data.frame()
+ 
+   for (pathway in pathways) {
+     if (pathway %in% names(data)) {
+       # Linear model with HI
+       model_hi <- lm(get(pathway) ~ Sex * HI + He, data = data)
+ 
+       # Linear model with He (heterozygosity)
+       model_he <- lm(get(pathway) ~ Sex * He + HI, data = data)
+ 
+       # Extract interaction p-values
+       hi_interaction_p <- summary(model_hi)$coefficients["SexM:HI", "Pr(>|t|)"]
+       he_interaction_p <- summary(model_he)$coefficients["SexM:He", "Pr(>|t|)"]
+ 
+       # Extract coefficients
+       hi_coef <- summary(model_hi)$coefficients["SexM:HI", "Estimate"]
+       he_coef <- summary(model_he)$coefficients["SexM:He", "Estimate"]
+ 
+       # Overall interaction test
+       model_full <- lm(get(pathway) ~ Sex * (HI + He), data = data)
+       anova_result <- anova(model_full)
+ 
+       # Store results
+       result_row <- data.frame(
+         Pathway = pathway,
+         Linear_coef = hi_coef,
+         Linear_interaction_p = hi_interaction_p,
+         Nonlinear_coef = he_coef,
+         Nonlinear_interaction_p = he_interaction_p,
+         Overall_interaction_p = min(hi_interaction_p, he_interaction_p, na.rm = TRUE),
+         stringsAsFactors = FALSE
+       )
+ 
+       results <- rbind(results, result_row)
+     }
+   }
+ 
+   # FDR correction
+   results$Overall_padj <- p.adjust(results$Overall_interaction_p, method = "BH")
+ 
+   return(results)
+ }
