# ==============================================================================
# COMPREHENSIVE RESULTS EXTRACTION FOR "HYBRID MALES SUFFER MORE" MANUSCRIPT
# ==============================================================================

# Load required libraries
library(tidyverse)
library(broom)

# Assuming field_mice is already loaded
# This script extracts all statistics needed for the Results section

# ==============================================================================
# 1. SAMPLE DESCRIPTION AND DATA COMPLETENESS
# ==============================================================================

cat("\n=== SAMPLE DESCRIPTION ===\n")

# Initial sample size
cat("Initial sample size:", nrow(field_mice), "\n")

# After removing missing HI
field_mice_complete <- field_mice %>% drop_na(HI)
cat("After removing missing HI:", nrow(field_mice_complete), "\n")

# Sex distribution
sex_table <- table(field_mice_complete$Sex)
cat("\nSex distribution:\n")
print(sex_table)

# Hybrid index summary
hi_summary <- field_mice_complete %>%
  summarise(
    n = n(),
    min = min(HI, na.rm = TRUE),
    q1 = quantile(HI, 0.25, na.rm = TRUE),
    median = median(HI, na.rm = TRUE),
    mean = mean(HI, na.rm = TRUE),
    q3 = quantile(HI, 0.75, na.rm = TRUE),
    max = max(HI, na.rm = TRUE),
    sd = sd(HI, na.rm = TRUE)
  )
cat("\nHybrid Index summary:\n")
print(hi_summary)

# Geographic extent
geo_summary <- field_mice_complete %>%
  summarise(
    n_locations = n_distinct(paste(Longitude, Latitude)),
    lat_min = min(Latitude, na.rm = TRUE),
    lat_max = max(Latitude, na.rm = TRUE),
    lon_min = min(Longitude, na.rm = TRUE),
    lon_max = max(Longitude, na.rm = TRUE)
  )
cat("\nGeographic summary:\n")
print(geo_summary)

# ==============================================================================
# 2. INFECTION STATUS ANALYSIS
# ==============================================================================

cat("\n=== INFECTION STATUS ===\n")

# Create complete infection status variable
field_mice_complete <- field_mice_complete %>%
  mutate(
    infection_status = case_when(
      MC.Eimeria == TRUE ~ "Infected",
      MC.Eimeria == FALSE ~ "Uninfected",
      TRUE ~ "Unknown"
    )
  )

# Infection status breakdown
infection_table <- table(field_mice_complete$infection_status, useNA = "ifany")
cat("\nInfection status:\n")
print(infection_table)

# Calculate percentage with known infection status
known_infection <- sum(field_mice_complete$infection_status != "Unknown")
pct_known <- round(known_infection / nrow(field_mice_complete) * 100, 1)
cat(sprintf("\nMice with known infection status: %d/%d (%.1f%%)\n", 
            known_infection, nrow(field_mice_complete), pct_known))

# Species identification among infected mice
species_summary <- field_mice_complete %>%
  filter(infection_status == "Infected") %>%
  count(species_Eimeria) %>%
  mutate(percentage = round(n / sum(n) * 100, 1))
cat("\nSpecies identification among infected mice:\n")
print(species_summary)

# Create dataset for complete case analysis
complete_data <- field_mice_complete %>%
  filter(infection_status != "Unknown")

cat(sprintf("\nComplete dataset for infection analysis: n = %d\n", nrow(complete_data)))

# Sex × Infection breakdown
sex_infection_table <- table(complete_data$Sex, complete_data$infection_status)
cat("\nSex × Infection status:\n")
print(sex_infection_table)

# ==============================================================================
# 3. PREDICTED WEIGHT LOSS ANALYSIS
# ==============================================================================

cat("\n=== PREDICTED WEIGHT LOSS ===\n")

# Overall distribution
pwl_summary <- field_mice_complete %>%
  summarise(
    n = n(),
    mean = mean(predicted_weight_loss, na.rm = TRUE),
    sd = sd(predicted_weight_loss, na.rm = TRUE),
    min = min(predicted_weight_loss, na.rm = TRUE),
    max = max(predicted_weight_loss, na.rm = TRUE)
  )
cat("\nOverall predicted weight loss:\n")
print(pwl_summary)

# Test for normality
shapiro_test <- shapiro.test(field_mice_complete$predicted_weight_loss)
cat("\nShapiro-Wilk normality test:\n")
cat(sprintf("W = %.3f, p = %.3f\n", shapiro_test$statistic, shapiro_test$p.value))

# By infection status (complete cases only)
pwl_by_infection <- complete_data %>%
  group_by(infection_status) %>%
  summarise(
    n = n(),
    mean = mean(predicted_weight_loss),
    sd = sd(predicted_weight_loss),
    se = sd / sqrt(n)
  )
cat("\nPredicted weight loss by infection status:\n")
print(pwl_by_infection)

# T-test for infection effect
t_test_infection <- t.test(predicted_weight_loss ~ infection_status, data = complete_data)
cat("\nT-test (Infected vs Uninfected):\n")
cat(sprintf("t(%0.1f) = %.2f, p = %.3f\n", 
            t_test_infection$parameter, 
            t_test_infection$statistic, 
            t_test_infection$p.value))
cat(sprintf("Mean difference = %.2f%% (95%% CI: %.2f to %.2f)\n",
            diff(t_test_infection$estimate),
            t_test_infection$conf.int[1],
            t_test_infection$conf.int[2]))

# Cohen's d
library(effsize)
cohen_d <- cohens_d(complete_data$predicted_weight_loss, 
                   complete_data$infection_status)
cohen_d
cat(sprintf("Cohen's d = %.3f\n", cohen_d$estimate))

# ==============================================================================
# 4. LINEAR MODEL ANALYSIS
# ==============================================================================

cat("\n=== LINEAR MODEL RESULTS ===\n")

# Simple model: infection effect
lm_simple <- lm(predicted_weight_loss ~ infection_status, data = complete_data)
cat("\nSimple model (infection only):\n")
print(tidy(lm_simple))

# Model with sex
lm_sex <- lm(predicted_weight_loss ~ infection_status + Sex, data = complete_data)
cat("\nModel with sex:\n")
print(tidy(lm_sex))

# Model with interaction
lm_interaction <- lm(predicted_weight_loss ~ infection_status * Sex, data = complete_data)
cat("\nModel with sex × infection interaction:\n")
print(tidy(lm_interaction))

# ==============================================================================
# 5. HYBRID INDEX DIFFERENCES BY INFECTION
# ==============================================================================

cat("\n=== HYBRID INDEX BY INFECTION STATUS ===\n")

# HI summary by infection status
hi_by_infection <- complete_data %>%
  group_by(infection_status) %>%
  summarise(
    n = n(),
    mean_HI = mean(HI),
    sd_HI = sd(HI),
    se_HI = sd_HI / sqrt(n)
  )
cat("\nHybrid index by infection status:\n")
print(hi_by_infection)

# T-test for HI differences
t_test_hi <- t.test(HI ~ infection_status, data = complete_data)
cat("\nT-test for HI (Infected vs Uninfected):\n")
cat(sprintf("t(%0.1f) = %.2f, p = %.3f\n", 
            t_test_hi$parameter, 
            t_test_hi$statistic, 
            t_test_hi$p.value))
cat(sprintf("Mean difference = %.3f (95%% CI: %.3f to %.3f)\n",
            diff(t_test_hi$estimate),
            t_test_hi$conf.int[1],
            t_test_hi$conf.int[2]))

# ==============================================================================
# 6. SUMMARY TABLE FOR MANUSCRIPT
# ==============================================================================

cat("\n=== SUMMARY TABLE ===\n")

# Create summary table
summary_table <- complete_data %>%
  group_by(Sex, infection_status) %>%
  summarise(
    n = n(),
    mean_pwl = mean(predicted_weight_loss),
    sd_pwl = sd(predicted_weight_loss),
    mean_HI = mean(HI),
    sd_HI = sd(HI),
    .groups = "drop"
  ) %>%
  arrange(Sex, infection_status)

cat("\nTable 1: Sample characteristics by sex and infection status\n")
print(summary_table, n = Inf)
